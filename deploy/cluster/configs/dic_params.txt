# =============================================================================
# DIC parameters for HPC cluster runs - MIRRORED mode (cppxdic_mirrored.sif)
# Copy and edit this file for each experiment. The container reads it at runtime via
# bind-mount (it is the file in WORKDIR): no rebuild needed for a change here.
#
# PATHS must use the CONTAINER mount points:
#   /data   -> your host DATA_DIR
#   /output -> your host OUTPUT_DIR
#
# FORMAT: one "key = value" per line. Do NOT put a comment after a value ("a,b #note"):
#   list values (param_filt_im, mirrored_pairs, ...) are rejected or misread. Put the
#   comment on its own line. A key that is not listed here is silently ignored.
#
# WHICH KEYS THE MIRRORED MODE USES (checked in the code on 2026-11-18, mirrored_mode.cpp
# and the StepDWorkflow functions it calls):
#   [mirrored: yes]  read by the mirrored mode.
#   [mirrored: NO]   read only by another mode (camerapairs = Steps E 3D reconstruction and
#                    F deformation, or singledic); changing it has NO effect on a mirrored
#                    run. Kept so that the same file still works for those modes.
#   untagged         not checked individually (general settings shared by the data import).
# The mirrored mode stops at the 2D DIC (Step D): no 3D reconstruction, no deformation.
# =============================================================================

# ---- 1. Paths (container mount points) --------------------------------------
base_path = /data
data_path = /data
dic_path = /output

# ---- 2. Subject / experiment ------------------------------------------------
subject_id = S09
phase_id = whole_slide
material_id = 1
ref_trial_id = 1
frictional_conditions = glass,coating,coating_oil
nfcond_set = 5
spddxlcond_set = 0.04,0.08
calib_folder_set = 5

# ---- 3. Sampling ------------------------------------------------------------
robot_sample_freq = 1000.0
vid_sample_freq = 50.0

# ---- 4. Frames (the CSV columns idx_frame_start/end/frame_jump override these) -
idx_frame_start = 16
idx_frame_end = 120
frame_jump = 1
# true: the three values above are authoritative; the protocol-derived per-phase
# window is not applied.
force_frame_window = true

# ---- 5. Image preprocessing -------------------------------------------------
im_filter_mode = true
# Grayscale saturation (satur.m clipping above limit_grayscale) on imported frames.
# Set false for videos already filtered/normalised upstream (e.g. MNG rigs): clipping
# such frames flattens the speckle contrast and makes ncorr diverge.
im_saturation_mode = true
# 0 = automatic (70 for subject numbers < 8, 100 otherwise); any positive value is
# used as-is (needed for subjects without a number).
limit_grayscale = 150
# [mirrored: yes] Band-pass of filter_like_ben (Matlab param_filt_im): radii r1,r2 in
# frequency pixels (r1 = high-pass, r2 = low-pass). Matlab default 25,300.
param_filt_im = 10,200
# [mirrored: yes] Grey-level percentiles low,high (%) used to normalise over the ROI.
# Default 5,95. After changing param_filt_im or filt_percentiles, delete the trial's
# mirrored_pair* folders, otherwise the cached ncorr results are reloaded.
filt_percentiles = 1,99

# ---- 6. Execution -----------------------------------------------------------
automatic_process = true
# [mirrored: yes] Tracking runs frame-parallel when true (the matching view 1 -> view 2
# is always sequential).
parallel_processing = true
debug_mode = false

# ---- 7. Mirrored-camera geometry (read ONLY by the mirrored build) ----------
# mirrored_num_pair: stereopairs 1..N exist (7 = 8 views -> 7 overlapping pairs).
#   Stereopair k -> views: 1:(1,2) 2:(2,3) 3:(3,4) 4:(4,5) 5:(6,5) 6:(7,6) 7:(8,7).
#   REF files use the stereopair number: REF_MASK_<ref trial>_<phase>_pair<k>.mat
#   A pair without REF_MASK/REF_SEED (here 2, 4, 6) fails alone and the run goes on.
mirrored_num_pair = 7
# mirrored_cam_order: physical camera of the view slots (views 1-2, 3-4, 5-6, 7-8).
#   Defaults 2,1,4,3 are those of the Manneback rig: VERIFY for DD4.
mirrored_cam_order = 2,1,4,3
# mirrored_pairs: stereopairs to process (1-based). Empty/absent = all 1..mirrored_num_pair.
mirrored_pairs = 1,3,5,7
# Write check_match_pair<N>.png per pair (ref | cur with ROI, seed, subset circle).
save_match_check_image = true

# ---- 8. Step D: 2D DIC tracking [mirrored: yes unless noted] ----------------
# Also the parameters of the matching view 1 -> view 2 (same spacing, threads, cutoffs),
# except the radius (step_e_radius below).
# [mirrored: NO] The mirrored mode always uses "regular".
step_d_analysis_type = regular
step_d_radius = 30
step_d_spacing = 10
step_d_cutoff_diffnorm = 1e-5
step_d_cutoff_iteration = 100
# Number of ncorr threads: must equal the --cpus-per-task of the job.
# Timing test 2026-11-18 (DD4 trial 3, spacing 20 and 10): 4 -> 16 threads only gains x1.5.
# Chosen 2026-11-18: 8 threads with --cpus-per-task=8 --mem=16G (MaxRSS 9.7 GB measured at 8 threads).
step_d_total_threads = 8
step_d_high_strain_enabled = true
step_d_seed_type = seed
step_d_auto_update = true
step_d_step_ref_change = 10
# [mirrored: yes] Applied when the cached DIC outputs are formatted (MATLAB
# step2_dic_finish): replaces badly correlated subsets by filtered values. Slow.
step_d_replacebadcorr = false

# ---- 9. Matching view 1 -> view 2 --------------------------------------------
# [mirrored: yes] Subset radius of the matching (MATLAB subset_radius_ncorr_matching).
step_e_radius = 40
# [mirrored: NO] Camerapairs only; mirrored takes step_d_cutoff_diffnorm.
step_e_cutoff_diffnorm = 1e-5

# ---- 10. Camerapairs-only settings (3D reconstruction, deformation) -----------
# [mirrored: NO] for every key of this section: the mirrored mode does not run Steps E/F.
# num_pair: number of stereopairs of the camerapairs mode (mirrored uses mirrored_num_pair).
num_pair = 4
# One entry per stereopair, in pair order (e.g. an 8-camera rig: 1,2;3,4;5,6;7,8).
camera_pairs = 1,2;3,4;5,6;7,8
# Remove distortion from the 2D points (MATLAB default: false).
step_e_distortion_removal = false
# Stitching of the per-pair 3D surfaces (when all pairs of a trial exist):
#   geometric = MATLAB-faithful overlap removal + boundary zipping (can take hours on
#               multi-pair rigs with large overlaps)
#   simple    = plain append of all pairs (overlaps kept), seconds
step_e_stitch_mode = geometric
# Accelerate the geometric stitcher's ray casts with an AABB hierarchy (exact: same
# stitched result, it only skips triangles the ray cannot reach).
step_e_ray_bvh = false
# Step F (deformation): temporal low-pass filter of the displacements, its cutoff (Hz),
# and the rigid-body-motion (RBM) / ARBM computation.
step_f_temporal_filtering = true
step_f_freq_filt = 10.0
step_f_compute_rbm = false

# ---- 11. Units & output -----------------------------------------------------
units_per_pixel = 0.2
subregion_radius = 20
fileversion = v1
data_format = mat
